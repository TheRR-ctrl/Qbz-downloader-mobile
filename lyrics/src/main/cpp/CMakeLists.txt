cmake_minimum_required(VERSION 3.18)
project(qlwhisper CXX C)
# whisper.cpp itself is fetched by the workflow (.github/workflows/whisper.yml) and given here as -DWHISPER_DIR=...
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_DIR} whisper.cpp)
add_library(qlwhisper SHARED qlwhisper.cpp)
target_link_libraries(qlwhisper whisper log)
target_link_options(qlwhisper PRIVATE -Wl,-z,max-page-size=16384)
